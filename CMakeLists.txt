  message(FATAL_ERROR "CLIENT_VERSION must be numeric major.minor.patch")
endif()

set(CMAKE_OBJCXX_STANDARD_REQUIRED ON)
set(CMAKE_AUTOMOC ON)

set(CMAKE_AUTOUIC ON)
if(MSVC)
  add_compile_options(/utf-8 /Zc:__cplusplus /permissive-)
endif()
set(CLIENT_QT_MAJOR 5 CACHE STRING "Qt major version (5 or 6)")
set(CLIENT_QT_HOST_BINS "" CACHE PATH "Matching native Qt 5 tools when cross-compiling macOS")
if(CLIENT_QT_HOST_BINS)
  if(NOT APPLE OR NOT CLIENT_QT_MAJOR STREQUAL "5")
    message(FATAL_ERROR "CLIENT_QT_HOST_BINS requires a macOS Qt 5 build")
  endif()
  foreach(tool qmake moc rcc uic)
    if(NOT EXISTS "${CLIENT_QT_HOST_BINS}/${tool}")
      message(FATAL_ERROR "Missing native Qt tool: ${tool}")
    endif()
    add_executable(Qt5::${tool} IMPORTED)
    set_target_properties(Qt5::${tool} PROPERTIES IMPORTED_LOCATION "${CLIENT_QT_HOST_BINS}/${tool}")
  endforeach()
endif()
if(CLIENT_QT_MAJOR STREQUAL "5")
  find_package(Qt5 5.15 REQUIRED COMPONENTS Core Gui Widgets Network Test Svg)
elseif(CLIENT_QT_MAJOR STREQUAL "6")
  find_package(Qt6 REQUIRED COMPONENTS Core Gui Widgets Network Test Svg)
else()
  message(FATAL_ERROR "CLIENT_QT_MAJOR must be 5 or 6")
endif()
set(QT_PACKAGE Qt${CLIENT_QT_MAJOR})
add_library(client_core STATIC src/appservice.cpp src/appservice.h src/platform.cpp src/platform.h)
target_include_directories(client_core PUBLIC src)
target_link_libraries(client_core PUBLIC ${QT_PACKAGE}::Core ${QT_PACKAGE}::Gui ${QT_PACKAGE}::Network)
target_compile_definitions(client_core PUBLIC QT_DISABLE_DEPRECATED_BEFORE=0x050F00 CLIENT_QT_VERSION="${PROJECT_VERSION}")
if(APPLE)
  target_link_libraries(client_core PRIVATE "-framework Security" "-framework CoreFoundation")
elseif(WIN32)
  target_link_libraries(client_core PRIVATE advapi32)
else()
  find_package(PkgConfig QUIET)
  if(PkgConfig_FOUND)
    pkg_check_modules(SECRET QUIET IMPORTED_TARGET libsecret-1)
  endif()
  if(SECRET_FOUND)
    target_link_libraries(client_core PRIVATE PkgConfig::SECRET)
    target_compile_definitions(client_core PRIVATE CLIENT_QT_HAS_LIBSECRET)
  else()
    message(WARNING "libsecret-1 missing: credentials will only live in memory; install libsecret-1-dev for persistent login")
  endif()
endif()

option(CLIENT_BUILD_LAUNCHERS "Build the existing Go native protocol launchers" ON)
if(CLIENT_BUILD_LAUNCHERS)
  find_package(Python3 3.8 REQUIRED COMPONENTS Interpreter)
  find_program(GO_EXECUTABLE go)
  if(NOT GO_EXECUTABLE)
    message(FATAL_ERROR "Go is required to build the protocol launchers")
  endif()
  if(APPLE)
    set(_goos darwin)
  elseif(WIN32)
    set(_goos windows)
  else()
    set(_goos linux)
  endif()
  string(TOLOWER "${CMAKE_SYSTEM_PROCESSOR}" _cpu)
  if(APPLE AND CMAKE_OSX_ARCHITECTURES)
    list(LENGTH CMAKE_OSX_ARCHITECTURES _arch_count)
    if(_arch_count GREATER 1)
      message(FATAL_ERROR "Build one architecture per package; universal launcher packaging must be assembled separately")
    endif()
    string(TOLOWER "${CMAKE_OSX_ARCHITECTURES}" _cpu)
  endif()
  if(_cpu MATCHES "^(arm64|aarch64)$")
    set(_goarch arm64)
  elseif(_cpu MATCHES "^(x86_64|amd64|x64)$")
    if(CMAKE_SIZEOF_VOID_P EQUAL 4)
      set(_goarch 386)
    else()
      set(_goarch amd64)
    endif()
  elseif(_cpu MATCHES "^(i[3-6]86|x86)$")
    set(_goarch 386)
  elseif(_cpu STREQUAL "loongarch64")
    set(_goarch loong64)
  elseif(_cpu STREQUAL "riscv64")
    set(_goarch riscv64)
  elseif(_cpu MATCHES "^(arm|armv[5-8].*)$")
    set(_goarch arm)
  elseif(_cpu STREQUAL "mips64el" OR _cpu STREQUAL "mips64le")
    set(_goarch mips64le)
  else()
    set(_goarch "")
  endif()
  set(CLIENT_GOOS "${_goos}" CACHE STRING "Go launcher target OS")
  set(CLIENT_GOARCH "${_goarch}" CACHE STRING "Go launcher target architecture")
  if(NOT CLIENT_GOARCH)
    message(FATAL_ERROR "Set CLIENT_GOARCH for target processor ${CMAKE_SYSTEM_PROCESSOR}")
  endif()
  add_custom_target(launchers ALL
    COMMAND ${Python3_EXECUTABLE} "${CMAKE_CURRENT_SOURCE_DIR}/scripts/build-launchers.py" --output "${CMAKE_CURRENT_BINARY_DIR}/bin" --go "${GO_EXECUTABLE}" --goos "${CLIENT_GOOS}" --goarch "${CLIENT_GOARCH}"
    WORKING_DIRECTORY "${CMAKE_CURRENT_SOURCE_DIR}" VERBATIM)
  add_dependencies(client-qt launchers)
  if(APPLE)
    add_custom_command(TARGET client-qt POST_BUILD
      COMMAND ${CMAKE_COMMAND} -E copy_directory "${CMAKE_CURRENT_BINARY_DIR}/bin" "$<TARGET_BUNDLE_DIR:client-qt>/Contents/Resources/bin")
    add_custom_target(bundle_launchers ALL
      COMMAND ${CMAKE_COMMAND} -E copy_directory "${CMAKE_CURRENT_BINARY_DIR}/bin" "$<TARGET_BUNDLE_DIR:client-qt>/Contents/Resources/bin"
      DEPENDS client-qt launchers VERBATIM)
  endif()
endif()

  find_package(Python3 3.8 REQUIRED COMPONENTS Interpreter)
  foreach(name appservice platform)
    add_executable(test_${name} tests/test_${name}.cpp resources/resources.qrc)
    target_link_libraries(test_${name} PRIVATE client_core ${QT_PACKAGE}::Test ${QT_PACKAGE}::Widgets)
    add_test(NAME ${name} COMMAND test_${name})
    set_tests_properties(${name} PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen" TIMEOUT 30)
  endforeach()
  target_compile_definitions(test_appservice PRIVATE
    TEST_PYTHON_EXECUTABLE="${Python3_EXECUTABLE}"
    TEST_TLS_PROXY="${CMAKE_CURRENT_SOURCE_DIR}/tests/tls_proxy.py")
  # Includes full login/refresh/revoke flows over real, untrusted TLS.
  # The full login/refresh/revoke flow over real TLS needs room on loaded runners.
  set_tests_properties(appservice PROPERTIES TIMEOUT 180)
  add_executable(test_ui tests/test_ui.cpp resources/resources.qrc)
  target_link_libraries(test_ui PRIVATE client_ui ${QT_PACKAGE}::Test)
  add_test(NAME ui COMMAND test_ui)
  set_tests_properties(ui PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen" TIMEOUT 30)
  add_test(NAME app_smoke COMMAND client-qt --smoke-test)
  set_tests_properties(app_smoke PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen" TIMEOUT 15)
  if(APPLE)
    add_executable(test_native_theme tests/test_native_theme.mm resources/resources.qrc)
    target_link_libraries(test_native_theme PRIVATE client_ui ${QT_PACKAGE}::Test "-framework AppKit")
    add_test(NAME native_theme COMMAND test_native_theme)
    set_tests_properties(native_theme PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen" TIMEOUT 15)
    add_executable(test_protocol_events tests/test_protocol_events.mm src/protocol_events.mm)
