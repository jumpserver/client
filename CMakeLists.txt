cmake_minimum_required(VERSION 3.16)
set(CLIENT_VERSION "4.1.4" CACHE STRING "Numeric installer version (major.minor.patch)")
if(NOT CLIENT_VERSION MATCHES "^[0-9]+\\.[0-9]+\\.[0-9]+$")
  message(FATAL_ERROR "CLIENT_VERSION must be numeric major.minor.patch")
endif()
project(client VERSION ${CLIENT_VERSION} LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_OBJCXX_STANDARD 17)
set(CMAKE_OBJCXX_STANDARD_REQUIRED ON)
set(CMAKE_AUTOMOC ON)
set(CMAKE_AUTORCC ON)
set(CMAKE_AUTOUIC ON)
if(MSVC)
  add_compile_options(/utf-8 /Zc:__cplusplus /permissive-)
endif()
set(CLIENT_QT_MAJOR 5 CACHE STRING "Qt major version (5 or 6)")
set(CLIENT_QT5_MIN_VERSION "5.15" CACHE STRING "Minimum Qt 5 version; use 5.12 for Kylin V10 SP1")
set(CLIENT_PACKAGE_VARIANT "" CACHE STRING "Optional Linux package name suffix, for example kylin")
set(CLIENT_QT_HOST_BINS "" CACHE PATH "Matching native Qt 5 tools when cross-compiling macOS")
if(CLIENT_QT_HOST_BINS)
  if(NOT APPLE OR NOT CLIENT_QT_MAJOR STREQUAL "5")
    message(FATAL_ERROR "CLIENT_QT_HOST_BINS requires a macOS Qt 5 build")
  endif()
  foreach(tool qmake moc rcc uic)
    if(NOT EXISTS "${CLIENT_QT_HOST_BINS}/${tool}")
      message(FATAL_ERROR "Missing native Qt tool: ${tool}")
    endif()
    add_executable(Qt5::${tool} IMPORTED)
    set_target_properties(Qt5::${tool} PROPERTIES IMPORTED_LOCATION "${CLIENT_QT_HOST_BINS}/${tool}")
  endforeach()
endif()
if(CLIENT_QT_MAJOR STREQUAL "5")
  find_package(Qt5 ${CLIENT_QT5_MIN_VERSION} REQUIRED COMPONENTS Core Gui Widgets Network Test Svg)
elseif(CLIENT_QT_MAJOR STREQUAL "6")
  find_package(Qt6 REQUIRED COMPONENTS Core Gui Widgets Network Test Svg)
else()
  message(FATAL_ERROR "CLIENT_QT_MAJOR must be 5 or 6")
endif()
set(QT_PACKAGE Qt${CLIENT_QT_MAJOR})
add_library(client_core STATIC src/appservice.cpp src/appservice.h src/platform.cpp src/platform.h src/networkworker.cpp src/networkworker.h)
target_include_directories(client_core PUBLIC src)
target_link_libraries(client_core PUBLIC ${QT_PACKAGE}::Core ${QT_PACKAGE}::Gui ${QT_PACKAGE}::Network)
if(CLIENT_QT_MAJOR STREQUAL "5" AND Qt5_VERSION VERSION_LESS 5.15)
  set(CLIENT_QT_DEPRECATED_BASELINE 0x050C00)
else()
  set(CLIENT_QT_DEPRECATED_BASELINE 0x050F00)
endif()
target_compile_definitions(client_core PUBLIC QT_DISABLE_DEPRECATED_BEFORE=${CLIENT_QT_DEPRECATED_BASELINE} CLIENT_QT_VERSION="${PROJECT_VERSION}")
if(APPLE)
  target_link_libraries(client_core PRIVATE "-framework Security" "-framework CoreFoundation" "-framework CFNetwork")
elseif(WIN32)
  target_link_libraries(client_core PRIVATE advapi32)
else()
  find_package(PkgConfig QUIET)
  if(PkgConfig_FOUND)
    pkg_check_modules(SECRET QUIET IMPORTED_TARGET libsecret-1)
  endif()
  if(SECRET_FOUND)
    target_link_libraries(client_core PRIVATE PkgConfig::SECRET)
    target_compile_definitions(client_core PRIVATE CLIENT_QT_HAS_LIBSECRET)
  else()
    message(WARNING "libsecret-1 missing: credentials will only live in memory; install libsecret-1-dev for persistent login")
  endif()
endif()
file(GLOB UI_SOURCES CONFIGURE_DEPENDS src/mainwindow.* src/connectiondialog.* src/settingsdialog.* src/theme*)
add_library(client_ui STATIC ${UI_SOURCES})
if(APPLE)
  enable_language(OBJCXX)
  target_sources(client_ui PRIVATE src/native_theme.mm src/native_theme.h)
  target_link_libraries(client_ui PRIVATE "-framework AppKit")
else()
  target_sources(client_ui PRIVATE src/native_theme.cpp src/native_theme.h)
endif()
target_link_libraries(client_ui PUBLIC client_core ${QT_PACKAGE}::Widgets ${QT_PACKAGE}::Svg)
add_executable(client WIN32 MACOSX_BUNDLE src/main.cpp resources/resources.qrc)
target_link_libraries(client PRIVATE client_ui)
set_target_properties(client PROPERTIES MACOSX_BUNDLE_BUNDLE_NAME "OSMClient" MACOSX_BUNDLE_GUI_IDENTIFIER "com.fit2cloud.osm.client" MACOSX_BUNDLE_BUNDLE_VERSION "${PROJECT_VERSION}" MACOSX_BUNDLE_SHORT_VERSION_STRING "${PROJECT_VERSION}" MACOSX_BUNDLE_ICON_FILE "icon.icns")
if(APPLE)
  enable_language(OBJCXX)
  target_sources(client PRIVATE src/protocol_events.mm src/protocol_events.h)
  target_link_libraries(client PRIVATE "-framework Foundation" "-framework AppKit")
  set_target_properties(client PROPERTIES MACOSX_BUNDLE_INFO_PLIST "${CMAKE_CURRENT_SOURCE_DIR}/packaging/Info.plist.in")
  target_sources(client PRIVATE resources/icon.icns)
  set_source_files_properties(resources/icon.icns PROPERTIES MACOSX_PACKAGE_LOCATION Resources)
elseif(WIN32)
  set_target_properties(client PROPERTIES OUTPUT_NAME "client")
  enable_language(RC)
  configure_file(packaging/client.rc.in "${CMAKE_CURRENT_BINARY_DIR}/client.rc" @ONLY)
  target_sources(client PRIVATE "${CMAKE_CURRENT_BINARY_DIR}/client.rc")
endif()
option(CLIENT_BUILD_LAUNCHERS "Build the existing Go native protocol launchers" ON)
if(CLIENT_BUILD_LAUNCHERS)
  find_package(Python3 3.8 REQUIRED COMPONENTS Interpreter)
  find_program(GO_EXECUTABLE go)
  if(NOT GO_EXECUTABLE)
    message(FATAL_ERROR "Go is required to build the protocol launchers")
  endif()
  if(APPLE)
    set(_goos darwin)
  elseif(WIN32)
    set(_goos windows)
  else()
    set(_goos linux)
  endif()
  string(TOLOWER "${CMAKE_SYSTEM_PROCESSOR}" _cpu)
  if(APPLE AND CMAKE_OSX_ARCHITECTURES)
    list(LENGTH CMAKE_OSX_ARCHITECTURES _arch_count)
    if(_arch_count GREATER 1)
      message(FATAL_ERROR "Build one architecture per package; universal launcher packaging must be assembled separately")
    endif()
    string(TOLOWER "${CMAKE_OSX_ARCHITECTURES}" _cpu)
  endif()
  if(_cpu MATCHES "^(arm64|aarch64)$")
    set(_goarch arm64)
  elseif(_cpu MATCHES "^(x86_64|amd64|x64)$")
    if(CMAKE_SIZEOF_VOID_P EQUAL 4)
      set(_goarch 386)
    else()
      set(_goarch amd64)
    endif()
  elseif(_cpu MATCHES "^(i[3-6]86|x86)$")
    set(_goarch 386)
  elseif(_cpu STREQUAL "loongarch64")
    set(_goarch loong64)
  elseif(_cpu STREQUAL "riscv64")
    set(_goarch riscv64)
  elseif(_cpu MATCHES "^(arm|armv[5-8].*)$")
    set(_goarch arm)
  elseif(_cpu STREQUAL "mips64el" OR _cpu STREQUAL "mips64le")
    set(_goarch mips64le)
  else()
    set(_goarch "")
  endif()
  set(CLIENT_GOOS "${_goos}" CACHE STRING "Go launcher target OS")
  set(CLIENT_GOARCH "${_goarch}" CACHE STRING "Go launcher target architecture")
  if(NOT CLIENT_GOARCH)
    message(FATAL_ERROR "Set CLIENT_GOARCH for target processor ${CMAKE_SYSTEM_PROCESSOR}")
  endif()
  add_custom_target(launchers ALL
    COMMAND ${Python3_EXECUTABLE} "${CMAKE_CURRENT_SOURCE_DIR}/scripts/build-launchers.py" --output "${CMAKE_CURRENT_BINARY_DIR}/bin" --go "${GO_EXECUTABLE}" --goos "${CLIENT_GOOS}" --goarch "${CLIENT_GOARCH}"
    WORKING_DIRECTORY "${CMAKE_CURRENT_SOURCE_DIR}" VERBATIM)
  add_dependencies(client launchers)
  if(APPLE)
    add_custom_command(TARGET client POST_BUILD
      COMMAND ${CMAKE_COMMAND} -E copy_directory "${CMAKE_CURRENT_BINARY_DIR}/bin" "$<TARGET_BUNDLE_DIR:client>/Contents/Resources/bin")
    add_custom_target(bundle_launchers ALL
      COMMAND ${CMAKE_COMMAND} -E copy_directory "${CMAKE_CURRENT_BINARY_DIR}/bin" "$<TARGET_BUNDLE_DIR:client>/Contents/Resources/bin"
      DEPENDS client launchers VERBATIM)
  endif()
endif()
install(TARGETS client BUNDLE DESTINATION . RUNTIME DESTINATION bin)
if(UNIX AND NOT APPLE)
  set(CMAKE_INSTALL_DEFAULT_DIRECTORY_PERMISSIONS OWNER_READ OWNER_WRITE OWNER_EXECUTE GROUP_READ GROUP_EXECUTE WORLD_READ WORLD_EXECUTE)
  install(DIRECTORY "${CMAKE_CURRENT_BINARY_DIR}/bin/" DESTINATION lib/client/bin USE_SOURCE_PERMISSIONS
          DIRECTORY_PERMISSIONS OWNER_READ OWNER_WRITE OWNER_EXECUTE GROUP_READ GROUP_EXECUTE WORLD_READ WORLD_EXECUTE OPTIONAL)
  install(FILES packaging/client.desktop DESTINATION share/applications)
  install(FILES resources/icon.png DESTINATION share/icons/hicolor/512x512/apps RENAME client.png)
  install(FILES LICENSE DESTINATION share/licenses/client)
elseif(WIN32)
  install(DIRECTORY "${CMAKE_CURRENT_BINARY_DIR}/bin/" DESTINATION bin/bin USE_SOURCE_PERMISSIONS OPTIONAL)
endif()

  include(packaging/linux.cmake)
